async function generatePDF() { try { const pdfBytes = await fetch('./Lifting%20Plan.pdf') .then(res => { if (!res.ok) throw new Error('ไม่พบไฟล์ Lifting Plan.pdf'); return res.arrayBuffer(); }); const pdfDoc = await PDFLib.PDFDocument.load(pdfBytes); const page = pdfDoc.getPages()[0]; const get = (id) => document.getElementById(id)?.value || ''; // ===== ข้อมูลทั่วไป ===== page.drawText(get('company'), { x: 140, y: 720, size: 12 }); page.drawText(get('brand'), { x: 95, y: 685, size: 12 }); page.drawText(get('model'), { x: 250, y: 685, size: 12 }); page.drawText(get('plate'), { x: 445, y: 685, size: 12 }); page.drawText(get('location'), { x: 140, y: 650, size: 12 }); page.drawText(get('project'), { x: 420, y: 650, size: 12 }); page.drawText(get('workdate'), { x: 510, y: 620, size: 12 }); page.drawText(get('driver'), { x: 140, y: 585, size: 12 }); page.drawText(get('signaler'), { x: 480, y: 585, size: 12 }); // ===== น้ำหนัก ===== page.drawText(get('mat'), { x: 600, y: 500, size: 12 }); page.drawText(get('rig'), { x: 600, y: 445, size: 12 }); page.drawText(get('total-field'), { x: 600, y: 390, size: 12 }); // ===== ผลคำนวณ ===== page.drawText(get('f-maxcrane'), { x: 600, y: 335, size: 12 }); page.drawText(get('f-radius'), { x: 600, y: 285, size: 12 }); page.drawText(get('f-boom'), { x: 600, y: 230, size: 12 }); page.drawText(get('f-cap'), { x: 600, y: 175, size: 12 }); const rate = document.getElementById('rate-box') .textContent .replace('%', ''); page.drawText(rate, { x: 510, y: 115, size: 12 }); // ===== Export ===== const output = await pdfDoc.save(); const blob = new Blob([output], { type: 'application/pdf' }); const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = 'LiftingPlan.pdf'; link.click(); } catch(err) { console.error(err); alert( 'Generate PDF ไม่สำเร็จ\n\n' + err.message ); } }
